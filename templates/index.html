try { // Идем на бэкенд за заказами const response = await fetch('/api/orders'); // Если сервер просит пароль (ошибка 401) if (response.status === 401 || response.status === 403) { main.innerHTML = ` <div class="flex flex-col items-center justify-center h-full"> <p class="text-gray-500 mb-4">Для просмотра заказов нужно войти в систему</p> <button class="bg-blue-600 text-white px-6 py-2 rounded-lg font-medium shadow-md hover:bg-blue-700">Войти</button> </div> `; return; } if (!response.ok) throw new Error('Ошибка сети'); const orders = await response.json(); // Если заказов нет if (!orders || orders.length === 0) { main.innerHTML = '<div class="flex items-center justify-center h-full"><p class="text-gray-500">У вас пока нет заказов</p></div>'; return; } // Рисуем карточки заказов со списком товаров let html = '<div class="space-y-4">'; orders.forEach(order => { // Сначала собираем HTML для списка товаров let itemsHtml = ''; if (order.items && order.items.length > 0) { itemsHtml = '<div class="mt-3 pt-3 border-t border-gray-100 space-y-2">'; order.items.forEach(item => { itemsHtml += ` <div class="flex justify-between text-sm"> <span class="text-gray-700">${item.name || item.material_code}</span> <span class="font-medium text-gray-900">${item.requested_quantity} ${item.unit}</span> </div> `; }); itemsHtml += '</div>'; } else { itemsHtml = '<p class="mt-3 pt-3 border-t border-gray-100 text-sm text-gray-400 italic">Нет товаров</p>'; } // Теперь собираем саму карточку html += ` <div class="bg-white p-4 rounded-xl shadow-sm border border-gray-100"> <div class="flex justify-between items-start mb-2"> <div> <h3 class="font-bold text-gray-800">Заказ #${order.id || '?'}</h3> <p class="text-xs text-gray-500 mt-1">${order.requesting_dept_id || 'Отдел не указан'} ➔ ${order.executing_dept_id || 'Склад не указан'}</p> </div> <span class="text-xs font-bold px-2 py-1 rounded-full bg-blue-100 text-blue-700">${order.status || 'Новый'}</span> </div> ${itemsHtml} </div> `; }); html += '</div>'; main.innerHTML = html; } catch (error) { main.innerHTML = '<div class="flex items-center justify-center h-full"><p class="text-red-500">Ошибка подключения к серверу</p></div>'; } } // Запускаем скрипт сразу после загрузки страницы loadApp(); </script>
